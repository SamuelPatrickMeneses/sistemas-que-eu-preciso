// 1. CAMADA DE VALIDAÇÃO PROGRAMÁTICA (Sanitizer)
// ==========================================
function validarEsanitizarDados(dadosBrutos) {
  // Regra 1: Garantir valores padrão para campos obrigatórios vazios
  const nome = dadosBrutos.nome ? dadosBrutos.nome.trim().toUpperCase() : "CANDIDATO";
  const cargo = dadosBrutos.cargo ? dadosBrutos.cargo.trim() : "Profissional";

  // Regra 2: Tratar strings longas de contato para não estourarem a largura da sidebar
  const contatoSanitizado = (dadosBrutos.contato || []).map(item => {
    let texto = item.trim();
    if (texto.length > 30) {
      return texto.substring(0, 27) + "..."; // Corta elegantemente se violar o layout físico
    }
    return texto;
  });

  // Regra 3: Limpar quebras de linha fantasmas comuns em colagens de texto
  const experienciasSanitizadas = (dadosBrutos.experiencias || []).map(exp => ({
    empresa: (exp.empresa || "Empresa").trim(),
    periodo: (exp.periodo || "Período").trim(),
    cargo: (exp.cargo || "Cargo").trim(),
    descricao: (exp.descricao || "").replace(/\s+/g, " ").trim() // Transforma múltiplos espaços/quebras em espaço único
  }));

  return {
    nome,
    cargo,
    contato: contatoSanitizado,
    competencias: dadosBrutos.competencias || [],
    experiencias: experienciasSanitizadas
  };
}
